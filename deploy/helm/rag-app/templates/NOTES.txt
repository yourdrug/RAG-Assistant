1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ (first $host.paths).path }}
{{- end }}
{{- else }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/component=server" -o jsonpath="{.items[0].metadata.name}")
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8001:8001
  Then visit: http://127.0.0.1:8001/health
{{- end }}

2. Check health:
  kubectl get pods -l "app.kubernetes.io/component=server"
  kubectl logs -l "app.kubernetes.io/component=server" --tail=50

3. Run migrations manually (if needed):
  kubectl exec -it deployment/{{ include "rag-app.fullname" . }}-server -- alembic upgrade head
